import requests
import json

OLLAMA_API = "http://localhost:11434/api/generate"
MODEL_NAME = "llama3:8b-instruct-q4_K_M"

SYSTEM_RULES = """You are a localized diagnostic terminal. 
Do not offer empathy or psychological analysis. 
When fed a behavioral error log, output one localized atomic commit (a single, rigid physical action) to interrupt the user's automated loop."""

def run_inference(prompt, system_prompt=SYSTEM_RULES):
    payload = {
        "model": MODEL_NAME,
        "system": system_prompt,
        "prompt": prompt,
        "stream": False,
        "options": {"temperature": 0.1}
    }
    response = requests.post(OLLAMA_API, json=payload)
    return response.json()["response"].strip()

def recursive_critic_loop(error_log):
    initial_prompt = f"Analyze this log and output a behavioral commit:\n{error_log}"
    draft_commit = run_inference(initial_prompt)

    critic_prompt = f"""Review your proposed commit against the core system rules.
    Original Log: {error_log}
    Proposed Commit: {draft_commit}
    
    Identify if the commit is too broad, attempts to solve multiple problems, or contains empathy. 
    Output only the strict, corrected atomic commit."""
    
    final_commit = run_inference(critic_prompt)

    # Currently logging as flat SFT. Needs to be converted to DPO Preference Pairs.
    dataset_entry = {
        "instruction": SYSTEM_RULES,
        "input": error_log,
        "output": final_commit
    }
    
    with open("unsloth_dataset.jsonl", "a") as file:
        file.write(json.dumps(dataset_entry) + "\n")

if __name__ == "__main__":
